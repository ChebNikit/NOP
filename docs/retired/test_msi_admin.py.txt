"""Regression checks for MSI property quoting and real Windows extraction."""
from hashlib import sha256
import os
from pathlib import Path, PureWindowsPath
import shutil
import subprocess
import tempfile
from types import SimpleNamespace
import unittest
from unittest import mock

import msi_admin

ROOT = Path(msi_admin.__file__).resolve().parent
FIXTURE = ROOT / 'tests/fixtures/msi-quote-test.msi'
FIXTURE_SHA256 = '2b54e73dfa035755c1ec30b54fd383e3f6901ba7ae94b16b7bc535c23758cd47'
PAYLOAD_SHA256 = '6b0559efc8ee6372ce4162d254dec51e0c9f741547a2b1e1cb48ffdaf3b2e808'


class MsiAdminTests(unittest.TestCase):
    def test_property_quotes_are_after_equals_for_the_reported_path(self):
        root = PureWindowsPath(r'C:\Users\Nikita\4 cnp CheNikPlayer\YamiFM_mpv_bundle\NOP_1.0\NOP_Installer_BuildKit_R11_UNVERIFIED (1)\NOP_BuildKit_Win10_11_R11')
        target = root / 'runtime_deps/converters/.libreoffice-123/unpacked'
        command = msi_admin.admin_command(r'C:\Windows\System32\msiexec.exe',
            root / '.build-tools/converter-downloads/LibreOffice.msi', target,
            root / 'logs/extract-libreoffice.log')
        self.assertIn(' TARGETDIR="' + str(target) + '" ', command)
        self.assertNotIn('"TARGETDIR=', command)
        self.assertIn('/a "' + str(root / '.build-tools/converter-downloads/LibreOffice.msi') + '"', command)

    def test_unicode_and_shell_characters_remain_literal(self):
        target = 'C:\\NOP \u044f (1) & %TEMP% ! ^\\unpacked'
        command = msi_admin.admin_command('C:\\Windows\\System32\\msiexec.exe',
            'C:\\NOP \u044f (1)\\source.msi', target, 'C:\\NOP \u044f (1)\\log.txt')
        self.assertIn(' TARGETDIR="' + target + '" ', command)

    def test_quote_and_control_characters_are_rejected(self):
        for char in ('"', '\0', '\r', '\n'):
            with self.subTest(char=repr(char)), self.assertRaises(ValueError):
                msi_admin.admin_command('msiexec.exe', 'source.msi', 'target' + char, 'log.txt')

    def invoke_fake_windows(self, temp, result=0, error=None):
        root = Path(temp)
        exe = root / 'Windows/System32/msiexec.exe'
        exe.parent.mkdir(parents=True, exist_ok=True)
        exe.touch()
        log = root / 'logs/extract.log'
        environment = SimpleNamespace(name='nt', environ={'WINDIR':str(root / 'Windows')})
        with mock.patch.object(msi_admin, 'os', environment), \
             mock.patch.object(msi_admin.subprocess, 'run',
                return_value=subprocess.CompletedProcess([], result), side_effect=error) as run:
            code = msi_admin.extract_msi(root/'source with spaces.msi', root/'target with spaces', log)
        return code, run, exe, log

    def test_exact_raw_command_is_passed_without_a_shell(self):
        with tempfile.TemporaryDirectory() as temp:
            code, run, exe, log = self.invoke_fake_windows(temp)
            self.assertEqual(code, 0)
            args, options = run.call_args
            self.assertIsInstance(args[0], str)
            self.assertFalse(options['shell'])
            self.assertEqual(options['executable'], str(exe))
            self.assertIn(' TARGETDIR="', args[0])
            self.assertEqual(log.with_suffix('.command.txt').read_text('utf-8-sig').strip(), args[0])

    def test_reboot_required_is_a_successful_extraction(self):
        with tempfile.TemporaryDirectory() as temp:
            code, _, _, _ = self.invoke_fake_windows(temp, result=3010)
            self.assertEqual(code, 3010)

    def test_native_error_keeps_code_and_command_log(self):
        with tempfile.TemporaryDirectory() as temp:
            with self.assertRaisesRegex(RuntimeError, '1639.*Invalid MSI command line'):
                self.invoke_fake_windows(temp, result=1639)
            self.assertTrue((Path(temp)/'logs/extract.command.txt').is_file())

    def test_native_timeout_is_reported(self):
        with tempfile.TemporaryDirectory() as temp:
            with self.assertRaisesRegex(RuntimeError, 'Cannot finish MSI extraction'):
                self.invoke_fake_windows(temp, error=subprocess.TimeoutExpired('msiexec.exe', 900))

    @unittest.skipUnless(os.name == 'nt', 'Windows Installer must run on Windows')
    def test_real_msi_extraction_in_unicode_paths(self):
        self.assertEqual(sha256(FIXTURE.read_bytes()).hexdigest(), FIXTURE_SHA256)
        with tempfile.TemporaryDirectory(prefix='nop-msi \u044f ') as temp:
            temp = Path(temp)
            source = temp/'source \u044f (1)'; source.mkdir()
            archive = source/'fixture \u044f.msi'
            shutil.copyfile(FIXTURE, archive)
            target = temp/'output \u044f (1)'
            msi_admin.extract_msi(archive, target, temp/'logs \u044f/fixture.log', timeout=90)
            files = list(target.rglob('payload.txt'))
            self.assertEqual(len(files), 1)
            self.assertEqual(sha256(files[0].read_bytes()).hexdigest(), PAYLOAD_SHA256)


if __name__ == '__main__':
    unittest.main()
