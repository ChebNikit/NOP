"""Extract private MSI packages using Windows Installer's property syntax."""
from pathlib import Path
import os
import subprocess


def admin_command(msiexec, archive, destination, log):
    # subprocess's default list serialization yields "TARGETDIR=C:\path with spaces".
    # MSI properties require TARGETDIR="C:\path with spaces" instead.
    values = tuple(map(str, (msiexec, archive, destination, log)))
    if any(any(char in value for char in ('"', '\0', '\r', '\n')) for value in values):
        raise ValueError('Unsupported character in an MSI extraction path')
    exe, package, target, logfile = values
    return (subprocess.list2cmdline([exe, '/a', package, '/qn', '/norestart']) +
            ' TARGETDIR="' + target + '" ' + subprocess.list2cmdline(['/L*V', logfile]))


def extract_msi(archive: Path, destination: Path, log: Path, timeout=900):
    if os.name != 'nt':
        raise RuntimeError('MSI administrative extraction requires Windows')
    msiexec = Path(os.environ['WINDIR']) / 'System32' / 'msiexec.exe'
    if not msiexec.is_file():
        raise RuntimeError('Windows Installer executable missing: ' + str(msiexec))
    archive, destination, log = archive.resolve(), destination.resolve(), log.resolve()
    destination.mkdir(parents=True, exist_ok=True)
    log.parent.mkdir(parents=True, exist_ok=True)
    command = admin_command(msiexec, archive, destination, log)
    command_log = log.with_suffix('.command.txt')
    command_log.write_text(command + '\n', encoding='utf-8-sig')
    print('Extracting private MSI: ' + archive.name, flush=True)
    print('MSI command: ' + command, flush=True)
    try:
        process = subprocess.run(command, executable=str(msiexec), shell=False,
            stdin=subprocess.DEVNULL, timeout=timeout,
            creationflags=getattr(subprocess, 'CREATE_NO_WINDOW', 0))
    except (OSError, subprocess.TimeoutExpired) as error:
        raise RuntimeError('Cannot finish MSI extraction. Command: ' + str(command_log) +
                           '; log: ' + str(log) + '; ' + str(error)) from error
    if process.returncode not in (0, 3010):
        detail = (' Invalid MSI command line.' if process.returncode == 1639 else '')
        raise RuntimeError(archive.name + ' MSI extraction failed: ' + str(process.returncode) +
                           '.' + detail + ' See ' + str(log) + '; command: ' + str(command_log))
    print('MSI extraction completed (exit ' + str(process.returncode) + ').', flush=True)
    return process.returncode
